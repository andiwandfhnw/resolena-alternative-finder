cmake_minimum_required(VERSION 3.14...3.28)
project(granite_schema_inspection LANGUAGES C CXX)

if(NOT DEFINED LLAMA_SOURCE_DIR OR NOT EXISTS "${LLAMA_SOURCE_DIR}/common/json-schema-to-grammar.h")
    message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to separately verified source commit 620d2225cc436ab0b001e3200e590e3b286ef9ab")
endif()

set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No dependency acquisition" FORCE)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "No dependency updates" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static inspection dependencies" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "Native schema converter" FORCE)
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "Pinned release label" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "No runtime tools" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "No server" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "No unified runtime" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "No examples" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "No upstream test suite" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "No UI build" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "No UI acquisition" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "No HTTPS dependency" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "No subprocess tools" FORCE)
set(LLAMA_LLGUIDANCE OFF CACHE BOOL "Pinned native GBNF route" FORCE)
set(GGML_CPU ON CACHE BOOL "Host-only model-free inspection" FORCE)
foreach(backend VULKAN CUDA HIP METAL SYCL OPENCL RPC)
    set(GGML_${backend} OFF CACHE BOOL "No accelerator/network backend for model-free inspection" FORCE)
endforeach()

add_subdirectory("${LLAMA_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama-source" EXCLUDE_FROM_ALL)
add_executable(granite-inspect-schema inspect-schema.cpp)
set_target_properties(granite-inspect-schema PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_compile_features(granite-inspect-schema PRIVATE cxx_std_17)
target_link_libraries(granite-inspect-schema PRIVATE llama-common)
